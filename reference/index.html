    def get_codec(program: str | Inputs) -> Codec:
    """Resolve the codec for a program name or input object.

    Resolution order: codecs passed to ``register``, built-ins, then codecs installed
    by other packages under the ``qccodec.codecs`` entry point group. A ``FileInput``
    resolves to the pass-through file codec.

    Args:
        program: A program name, or an input whose program should be resolved.

    Raises:
        CodecNotFoundError: No codec is available for the program.
    """
    if isinstance(program, ProgramInput):
        name = program.program
    elif isinstance(program, FileInput):
        from .programs.file import codec as file_codec

        return file_codec
    else:
        name = program

    codec = _REGISTERED.get(name)
    if codec is None and name in _BUILTIN:
        module = import_module(f"qccodec.programs.{_BUILTIN[name]}")
        codec = module.codecs[name] if hasattr(module, "codecs") else module.codec
    if codec is None:
        codec = _plugin(name)
    if codec is None:
        raise CodecNotFoundError(f"No codec found for program '{name}'.")
    return codec

    self,
    program: str,

    def decoder(self, func: F) -> F:
    """Register ``func(input_data, native_output) -> Data`` in place of parsers.

    Raises:
        RegistryError: The codec has parsers or options, which only the parser
            pipeline uses.
    """
    if self.parsers:
        raise RegistryError(
            f"Codec {self.program!r} has parsers; a decoder would replace them."
        )
    if self.options:
        raise RegistryError(
            f"Codec {self.program!r} declares options, which only parsers receive."
        )
    self._decode = func
    return func

    def parser(
    self,
    *,
    target: Target | None = None,
    calctypes: Collection[CalcType | str] | None = None,
    file: FileRef = STDOUT,
    required: bool = True,
    versions: str | None = None,
) -> Callable[[F], F]:
    """Register a parser function.

    Args:
        target: Result field to fill, or a tuple for a nested field. Directory
            parsers may omit it and return a mapping merged into the result.
        calctypes: Calculation types this parser applies to; all when omitted.
        file: ``"stdout"`` (default), ``"directory"`` (the parser receives a
            ``Context``), a file name/glob relative to the directory, or a
            function of the ``Context`` returning such a name.
        required: If True, a missing file or ``MatchNotFoundError`` fails
            decoding unless partial data is being recovered.
        versions: Program versions this parser applies to, as a PEP 440
            specifier such as ``">=6.0"`` or ``">=5.0,<6.0"``; every version
            when omitted. Parsers for the same target may coexist when each
            declares ``versions``.

    Returns:
        The original function, unchanged, so it can be called and tested directly.
    """
    types = (
        frozenset(CalcType)
        if calctypes is None
        else frozenset(CalcType(c) for c in calctypes)
    )
    try:
        specifier = SpecifierSet(versions) if versions is not None else None
    except InvalidSpecifier as exc:
        raise RegistryError(f"Invalid versions specifier {versions!r}.") from exc

    def decorate(func: F) -> F:
        """Validate and record the parser specification."""
        if self._decode is not None:
            raise RegistryError(
                f"Codec {self.program!r} has a decoder; parsers would never run."
            )
        if target is None and file != DIRECTORY:
            raise RegistryError(
                f"Parser '{func.__name__}' must have a target unless file='directory'."
            )
        if target is not None and not (
            isinstance(target, str)
            or (
                isinstance(target, tuple)
                and all(isinstance(k, str) for k in target)
            )
        ):
            raise RegistryError(
                f"Target {target!r} must be a string or tuple of strings."
            )
        if specifier is not None and target == VERSION_TARGET:
            raise RegistryError(
                "The program version parser runs on every version; "
                "it cannot declare versions."
            )
        for existing in self.parsers:
            if (
                target is not None
                and existing.target == target
                and types & existing.calctypes
                and (specifier is None or existing.versions is None)
            ):
                raise RegistryError(
                    f"Duplicate parser target {target!r} for overlapping "
                    "calculation types. Declare versions on both parsers if "
                    "they handle different program versions."
                )
        self.parsers.append(
            ParserSpec(func, file, types, required, target, specifier)
        )
        return func

    return decorate

    def encode(self, input_data: Inputs) -> NativeInput:
    """Translate an input into a native invocation without executing it.

    Raises:
        EncoderError: The codec has no encoder, the input's program or
            calculation type is not supported, or the encoder returned
            something other than a ``SubprocessInput`` or ``PythonInput``.
    """
    if self._encode is None:
        raise EncoderError(f"Codec {self.program!r} has no encoder.")
    if isinstance(input_data, ProgramInput):
        if input_data.program != self.program:
            raise EncoderError(
                f"Codec {self.program!r} cannot encode {input_data.program!r}."
            )
        if input_data.calctype not in self.calctypes:
            raise EncoderError(
                f"{self.program} does not support {input_data.calctype.value}."
            )
    native = self._encode(input_data)
    if not isinstance(native, (SubprocessInput, PythonInput)):
        raise EncoderError(
            f"{self.program} encoder must return SubprocessInput or PythonInput, "
            f"not {type(native).__name__}."
        )
    return native

    def decode(
    self,
    input_data: Inputs,
    native_output: NativeOutput,
    options: Mapping[str, Any] | None = None,
) -> Data:
    """Interpret native output, returning data or raising with partial data.

    When the run failed (for example a nonzero exit status), whatever can be
    parsed is recovered, and if the codec's failure hook explains the failure
    ``CalculationError`` carries that explanation and the recovered data.

    Args:
        input_data: The input that produced the output.
        native_output: What the runner observed.
        options: Overrides for this codec's declared ``options``.

    Raises:
        CalculationError: The program failed and the failure hook says why.
        ParserError: A required output is missing or malformed in a run that
            did not fail.
    """
    if self._decode is not None:
        self._merge_options(options)  # rejects unknown keys
        return self._decode(input_data, native_output)
    if not isinstance(input_data, ProgramInput):
        raise ParserError(f"Codec {self.program!r} has no decoder for file inputs.")
    return self.parse(
        input_data.calctype,
        stdout=native_output.stdout,
        directory=native_output.directory,
        input_data=input_data,
        failed=native_output.failed,
        options=options,
        report_failure=True,
    )

    def parse(
    self,
    calctype: CalcType | str,
    *,
    stdout: str | None = None,
    directory: str | Path | None = None,
    input_data: ProgramInput | None = None,
    failed: bool = False,
    options: Mapping[str, Any] | None = None,
    version: str | None = None,
    as_dict: bool = False,
    check_failure: bool = True,
    report_failure: bool | None = None,
) -> StructuredData | dict[str, Any]:
    """Run this codec's parsers over saved artifacts without running the program.

    Args:
        calctype: The type of calculation that produced the artifacts.
        stdout: The program's captured output.
        directory: The calculation directory holding output files.
        input_data: The input used for the calculation, for parsers that need it.
        failed: Recover what is available from a failed calculation. Missing
            files and values are tolerated; malformed values still fail.
        options: Overrides for this codec's declared ``options``.
        version: The program version, when the output does not print it or
            should be overridden. Otherwise the version parser supplies it.
        as_dict: Return the collected values instead of a validated model.
        check_failure: Run the codec's failure hook first. Nested decodes of
            partial logs turn this off.
        report_failure: Raise ``CalculationError`` when the failure hook
            explains a failure. Defaults to ``not failed``, so recovery mode
            returns partial data quietly; ``decode`` passes True so a program
            that exited with an error still reports why.

    Returns:
        The qcdata model for ``calctype``, or a dictionary when ``as_dict``.

    Raises:
        ParserError: A required file is missing or a value is malformed.
        MatchNotFoundError: A required parser found no match.
        CalculationError: The failure hook explained a failure and
            ``report_failure`` is true. Recovered data is attached to every
            exception as ``.data``.
    """
    ctx = Context(
        codec=self,
        calctype=CalcType(calctype),
        stdout=stdout,
        directory=Path(directory) if directory is not None else None,
        input_data=input_data,
        failed=failed,
        options=self._merge_options(options),
    )
    data_type = get_data_type(ctx.calctype)
    message = (
        self._failure(ctx) if self._failure is not None and check_failure else None
    )
    if message:
        ctx.failed = True
    collected: dict[str, Any] = {"provenance": {"program": self.program}}
    not_found: MatchNotFoundError | None = None
    missing: list[str] = []
    # With an unknown version, version-gated alternatives for one target are
    # tried in order; a miss only counts if no alternative fills the target.
    deferred: dict[object, MatchNotFoundError | str] = {}
    filled: set[object] = set()

    def partial() -> StructuredData:
        """Validate whatever has been collected so far."""
        return data_type(**collected)

    specs = [s for s in self.parsers if ctx.calctype in s.calctypes]
    if version is not None:
        ctx.version = version
        _assign(collected, VERSION_TARGET, version)
        specs = [s for s in specs if s.target != VERSION_TARGET]
    else:
        # The version parser runs first so later parsers can be filtered by it.
        specs.sort(key=lambda s: s.target != VERSION_TARGET)
    for spec in specs:
        name = spec.parser.__name__
        key: object = spec.target if spec.target is not None else name
        gated = spec.versions is not None
        matched = self._version_matches(spec, ctx.version)
        if matched is False:
            continue
        if matched is None and key in filled:
            continue  # an earlier alternative already handled this target
        try:
            value: Any = spec.parser(ctx.contents(spec))
        except FileNotFoundError as exc:
            logger.info("Parser %r: missing %s", name, exc.filename or exc)
            if spec.required and not ctx.failed:
                if matched is None and gated:
                    deferred.setdefault(key, str(exc.filename or exc))
                else:
                    missing.append(str(exc.filename or exc))
            continue
        except MatchNotFoundError as exc:
            logger.info("Parser %r found no match", name)
            if spec.required and not ctx.failed:
                if matched is None and gated:
                    deferred.setdefault(key, exc)
                elif not_found is None:
                    not_found = exc
            continue
        except ParserError as exc:
            exc.data = partial()
            raise
        if spec.target is None:
            for item_key, item in value.items():
                _assign(collected, item_key, item)
        else:
            _assign(collected, spec.target, value)
        if spec.target == VERSION_TARGET:
            ctx.version = value
        filled.add(key)
        deferred.pop(key, None)
        logger.debug("Parser %r filled %r", name, spec.target)

    for problem in deferred.values():
        if isinstance(problem, MatchNotFoundError):
            not_found = not_found or problem
        else:
            missing.append(problem)

    if not_found is not None:
        not_found.data = partial()
        raise not_found
    if missing:
        raise ParserError(
            "Missing required output artifact(s): "
            + ", ".join(sorted(set(missing))),
            data=partial(),
        )
    if report_failure is None:
        report_failure = not failed
    if message and report_failure:
        raise CalculationError(message, data=partial())
    return collected if as_dict else partial()

    def path(self, pattern: str) -> Path:
    """Resolve a file name or glob pattern to exactly one path in the directory.

    Raises:
        FileNotFoundError: No directory is available or nothing matches.
        ParserError: The pattern matches more than one file.
    """
    if self.directory is None:
        raise FileNotFoundError(pattern)
    matches = sorted(p for p in self.directory.glob(pattern) if p.is_file())
    if not matches:
        raise FileNotFoundError(pattern)
    if len(matches) > 1:
        names = ", ".join(str(p.relative_to(self.directory)) for p in matches)
        raise ParserError(f"Pattern {pattern!r} matched several files: {names}")
    return matches[0]

    Testing codecs¶

    Test helpers for codecs: fixture cases, a case runner, and a codec contract check.

    A case is a directory holding one program run and what a codec must make of it::

    tests/data/<program>/<case>/
    input.json      ProgramInput used for the run (program, calctype, structure, ...)
    expected.json   the qcdata *Data object the codec must produce
    stdout          captured program output (optional)
    <files>         output files at their real relative paths, e.g. scr.geom/optim.xyz
    case.json       optional settings: {"options": {...}, "failed": true,
                    "raises": "MatchNotFoundError", "directory": true}
    encoded/        optional: command.txt (one argument per line) and the native
                    input files codec.encode(input) must generate

    cases(root) discovers every such directory below root; run_case checks encoding, the full decode, and each registered parser on its own file against the matching slice of expected.json; check_codec verifies the structural contract every codec must satisfy, with or without fixtures.

    Typical test module::

    import pytest
from qccodec import get_codec
from qccodec.testing import cases, run_case

@pytest.mark.parametrize("case", cases("tests/data"), ids=str)
def test_case(case):
    run_case(get_codec(case.input_data.program), case)

    def check_parsers(
    codec: Codec, case: Case, *, rtol: float = 1e-9, atol: float = 0.0
) -> None:
    """Call every applicable parser directly and compare with its expected slice."""
    assert case.expected is not None
    expected = _normalize(case.expected)
    ctx = Context(
        codec=codec,
        calctype=case.input_data.calctype,
        stdout=case.stdout,
        directory=case.directory,
        input_data=case.input_data,
        failed=case.failed,
        options={**codec.options, **case.options},
    )
    for spec in codec.parsers:
        if ctx.calctype not in spec.calctypes:
            continue
        try:
            contents = ctx.contents(spec)
        except FileNotFoundError:
            continue
        where = f"{case} parser {spec.parser.__name__}"
        try:
            value = spec.parser(contents)
        except MatchNotFoundError:
            assert not spec.required or case.failed, (
                f"{where}: required parser found no match"
            )
            if spec.target is not None:
                found = _expected_slice(expected, spec.target)
                assert found in (None, [], {}), (
                    f"{where}: no match, but expected.json holds {found!r}"
                )
            continue
        if spec.target is None:
            for key, item in value.items():
                assert_data_equal(
                    item,
                    expected.get(key),
                    rtol=rtol,

                value,

    def run_case(
    codec: Codec, case: Case, *, rtol: float = 1e-9, atol: float = 0.0
) -> None:
    """Check a codec against one fixture case.

    Verifies, in order: the encoded native inputs when ``encoded/`` exists; the full
    decode, which must produce ``expected.json`` or raise the exception named in
    ``case.json`` with ``expected.json`` attached as partial data; and each registered
    parser applied directly to its own file.

    Raises:
        AssertionError: With a description of the first mismatch.
    """
    if case.encoded is not None:
        check_encoding(codec, case, rtol=rtol)
    kwargs: dict[str, Any] = dict(
        stdout=case.stdout,
        directory=case.directory,
        input_data=case.input_data,
        failed=case.failed,
        options=case.options or None,
    )
    if case.raises:
        exc_type = _exception_class(case.raises)
        try:
            codec.parse(case.input_data.calctype, **kwargs)
        except exc_type as exc:
            if case.expected is not None:
                assert_data_equal(
                    getattr(exc, "data", None),
                    case.expected,
                    rtol=rtol,
                    atol=atol,
                    where=f"{case} partial",
                )
        else:
            raise AssertionError(f"{case}: expected {case.raises} but decode succeeded")
        return
    assert case.expected is not None, (
        f"{case}: expected.json is required unless raises is set"
    )
    data = codec.parse(case.input_data.calctype, **kwargs)
    assert_data_equal(data, case.expected, rtol=rtol, atol=atol, where=str(case))
    if codec.parsers:
        check_parsers(codec, case, rtol=rtol, atol=atol)

    def check_codec(
    codec: Codec,
    *,
    inputs: Iterable[ProgramInput] = (),
    calctypes: Collection[CalcType] | None = None,
) -> None:
    """Verify the structural contract every codec must satisfy.

    Checks that parser targets name real result fields for every calculation type
    they apply to, that decoding nothing in recovery mode yields a typed empty result
    and in strict mode raises with typed partial data, that option defaults are
    well formed, and that encoding each of ``inputs`` yields a native invocation whose
    files stay inside the calculation directory.

    Args:
        codec: The codec to check.
        inputs: Inputs the codec must be able to encode.
        calctypes: Calculation types to exercise; defaults to ``codec.calctypes``.

    Raises:
        AssertionError: Listing every problem found.
    """
    problems: list[str] = []
    types = list(calctypes if calctypes is not None else codec.calctypes)
    if not codec.program:
        problems.append("program name is empty")
    if not codec.calctypes:
        problems.append("no calctypes declared")
    for key in codec.options:
        if not isinstance(key, str) or not key:
            problems.append(f"option name {key!r} is not a non-empty string")

    for spec in codec.parsers:
        problems.extend(_check_spec(spec, types))

    if codec.parsers:
        for calctype in types:
            data_type = get_data_type(calctype)
            try:
                empty = codec.parse(calctype, failed=True)
            except Exception as exc:  # noqa: BLE001
                problems.append(f"{calctype.value}: parse(failed=True) raised {exc!r}")
            else:
                if not isinstance(empty, data_type):
                    problems.append(
                        f"{calctype.value}: parse(failed=True) returned {type(empty).__name__}"
                    )
            try:
                codec.parse(calctype)
            except ParserError as exc:
                if not isinstance(exc.data, data_type):
                    problems.append(
                        f"{calctype.value}: strict parse raised without typed partial data"
                    )
            except Exception as exc:  # noqa: BLE001
                problems.append(f"{calctype.value}: strict parse raised {exc!r}")

    for inp in inputs:
        try:
            native = codec.encode(inp)
        except Exception as exc:  # noqa: BLE001
            problems.append(f"encode({inp.calctype.value}) raised {exc!r}")
            continue
        for name in native.files:
            if Path(name).is_absolute() or ".." in Path(name).parts:
                problems.append(
                    f"encode({inp.calctype.value}) file escapes directory: {name!r}"
                )
        if isinstance(native, SubprocessInput) and not native.command:
            problems.append(f"encode({inp.calctype.value}) returned an empty command")
        if isinstance(native, PythonInput) and not callable(native.function):
            problems.append(f"encode({inp.calctype.value}) function is not callable")

    gated = [
        spec.parser.__name__ for spec in codec.parsers if spec.versions is not None
    ]
    has_version_parser = any(spec.target == VERSION_TARGET for spec in codec.parsers)
    if gated and not has_version_parser:
        problems.append(
            f"parsers {gated} declare versions but no parser fills "
            f"{VERSION_TARGET!r}, so the version is only known when passed in"
        )

    if codec.propagate_wfn is not None and not callable(codec.propagate_wfn):
        problems.append("propagate_wfn is not callable")
    if problems:
        raise AssertionError(
            f"Codec {codec.program!r} failed checks:\n  " + "\n  ".join(problems)
        )
